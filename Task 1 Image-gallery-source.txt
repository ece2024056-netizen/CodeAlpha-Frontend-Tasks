<script>
  const photos = [
    { seed: "ridge-line",       title: "Ridge Line",       cat: "nature",   src: "images/ridge-line.jpg" },
    { seed: "valley-fog",       title: "Low Valley Fog",   cat: "nature",   src: "images/valley-fog.jpg" },
    { seed: "coast-break",      title: "Coast Break",      cat: "nature",   src: "images/coast-break.jpg" },
    { seed: "pine-corridor",    title: "Pine Corridor",    cat: "nature",   src: "images/pine-corridor.jpg" },
    { seed: "downtown-grid",    title: "Downtown Grid",    cat: "urban",    src: "images/downtown-grid.jpg" },
    { seed: "concrete-stair",   title: "Concrete Stair",   cat: "urban",    src: "images/concrete-stair.jpg" },
    { seed: "night-signal",     title: "Night Signal",     cat: "urban",    src: "images/night-signal.jpg" },
    { seed: "overpass",         title: "Overpass",         cat: "urban",    src: "images/overpass.jpg" },
    { seed: "quiet-study",      title: "Quiet Study",      cat: "portrait", src: "images/quiet-study.jpg" },
    { seed: "window-light",     title: "Window Light",     cat: "portrait", src: "images/window-light.jpg" },
    { seed: "half-turn",        title: "Half-turn",        cat: "portrait", src: "images/half-turn.jpg" },
    { seed: "station-bench",    title: "Station Bench",    cat: "portrait", src: "images/station-bench.jpg" },
    { seed: "departure-gate",   title: "Departure Gate",   cat: "travel",   src: "images/departure-gate.jpg" },
    { seed: "market-road",      title: "Market Road",      cat: "travel",   src: "images/market-road.jpg" },
    { seed: "mountain-pass",    title: "Mountain Pass",    cat: "travel",   src: "images/mountain-pass.jpg" },
    { seed: "harbor-town",      title: "Harbor Town",      cat: "travel",   src: "images/harbor-town.jpg" },
  ];

  const grid = document.getElementById('grid');

  function frameLabel(n){
    return 'FR·' + String(n).padStart(2,'0');
  }

  // ---------- deterministic pseudo-random from a seed string ----------
  function hash(str){
    let h = 0;
    for (let i = 0; i < str.length; i++){ h = (h * 31 + str.charCodeAt(i)) >>> 0; }
    return h;
  }
  function rand(seed, i){
    const x = Math.sin(hash(seed) + i * 999) * 10000;
    return x - Math.floor(x);
  }

  const PALETTE = {
    nature:   { a: '#3c4a2f', b: '#6b7255', c: '#e8a33d' },
    urban:    { a: '#22201a', b: '#423c30', c: '#b23a24' },
    portrait: { a: '#2c261c', b: '#6b5a3d', c: '#e8a33d' },
    travel:   { a: '#2a2f3c', b: '#5a6a7a', c: '#b23a24' },
  };

  // Procedural "exposure" art per category — no network dependency, purely vector.
  function makeArt(p, w, h){
    if (p.src){
      return `<img src="${p.src}" alt="${p.title}" loading="lazy" decoding="async">`;
    }
    const pal = PALETTE[p.cat];
    const r1 = rand(p.seed, 1), r2 = rand(p.seed, 2), r3 = rand(p.seed, 3), r4 = rand(p.seed, 4);
    let shapes = '';

    if (p.cat === 'nature'){
      const horizon = h * (0.55 + r1 * 0.15);
      shapes += `<rect x="0" y="0" width="${w}" height="${horizon}" fill="url(#sky-${p.seed})"/>`;
      shapes += `<rect x="0" y="${horizon}" width="${w}" height="${h-horizon}" fill="${pal.a}"/>`;
      shapes += `<circle cx="${w*(0.2+r2*0.6)}" cy="${horizon*0.5}" r="${w*0.06}" fill="${pal.c}" opacity="0.85"/>`;
      for (let i=0;i<4;i++){
        const rx = w*rand(p.seed,10+i);
        const ry = horizon - i*8 - rand(p.seed,20+i)*10;
        shapes += `<polygon points="${rx-40},${horizon} ${rx},${ry-70} ${rx+40},${horizon}" fill="${pal.b}" opacity="${0.5+i*0.1}"/>`;
      }
    } else if (p.cat === 'urban'){
      shapes += `<rect x="0" y="0" width="${w}" height="${h}" fill="${pal.a}"/>`;
      const cols = 6, rows = 8;
      for (let cx=0; cx<cols; cx++){
        const bw = w/cols;
        const bh = h * (0.4 + rand(p.seed, cx+30)*0.55);
        shapes += `<rect x="${cx*bw+4}" y="${h-bh}" width="${bw-8}" height="${bh}" fill="${pal.b}" opacity="0.9"/>`;
        for (let ry=0; ry<rows; ry++){
          if (rand(p.seed, cx*20+ry) > 0.55){
            const wy = h - bh + 14 + ry*(bh/rows);
            if (wy < h-10) shapes += `<rect x="${cx*bw+12}" y="${wy}" width="10" height="8" fill="${pal.c}" opacity="0.8"/>`;
          }
        }
      }
    } else if (p.cat === 'portrait'){
      shapes += `<rect x="0" y="0" width="${w}" height="${h}" fill="${pal.a}"/>`;
      shapes += `<ellipse cx="${w/2}" cy="${h*0.42}" rx="${w*0.24}" ry="${h*0.3}" fill="url(#glow-${p.seed})"/>`;
      shapes += `<ellipse cx="${w*(0.4+r3*0.2)}" cy="${h*0.9}" rx="${w*0.4}" ry="${h*0.3}" fill="${pal.b}" opacity="0.5"/>`;
    } else {
      shapes += `<rect x="0" y="0" width="${w}" height="${h}" fill="url(#sky-${p.seed})"/>`;
      const horizon = h * (0.6 + r4*0.1);
      shapes += `<rect x="0" y="${horizon}" width="${w}" height="${h-horizon}" fill="${pal.b}"/>`;
      shapes += `<polygon points="${w*0.15},${horizon} ${w*0.35},${horizon-h*0.22} ${w*0.55},${horizon}" fill="${pal.a}"/>`;
      shapes += `<polygon points="${w*0.45},${horizon} ${w*0.68},${horizon-h*0.15} ${w*0.9},${horizon}" fill="${pal.a}" opacity="0.85"/>`;
      shapes += `<circle cx="${w*0.78}" cy="${horizon*0.35}" r="${w*0.045}" fill="${pal.c}"/>`;
    }

    // fine grain speckle for a film feel
    let grain = '';
    for (let i=0;i<40;i++){
      const gx = rand(p.seed, 100+i)*w, gy = rand(p.seed, 200+i)*h;
      grain += `<circle cx="${gx}" cy="${gy}" r="0.7" fill="#000" opacity="${0.04+rand(p.seed,300+i)*0.05}"/>`;
    }

    return `
      <svg viewBox="0 0 ${w} ${h}" xmlns="http://www.w3.org/2000/svg" preserveAspectRatio="xMidYMid slice">
        <defs>
          <linearGradient id="sky-${p.seed}" x1="0" y1="0" x2="0" y2="1">
            <stop offset="0%" stop-color="${pal.c}" stop-opacity="0.55"/>
            <stop offset="100%" stop-color="${pal.a}"/>
          </linearGradient>
          <radialGradient id="glow-${p.seed}" cx="50%" cy="35%" r="65%">
            <stop offset="0%" stop-color="${pal.c}" stop-opacity="0.7"/>
            <stop offset="100%" stop-color="${pal.a}" stop-opacity="0"/>
          </radialGradient>
        </defs>
        ${shapes}
        ${grain}
      </svg>`;
  }

  photos.forEach((p, i) => {
    const cell = document.createElement('button');
    cell.type = 'button';
    cell.className = 'cell';
    cell.dataset.cat = p.cat;
    cell.dataset.index = i;
    cell.setAttribute('aria-label', `Open ${p.title}, ${p.cat}, ${frameLabel(i + 1)}`);
    cell.style.animationDelay = (i * 0.03) + 's';
    cell.innerHTML = `
      <span class="frame-tag">${frameLabel(i+1)}</span>
      <span class="art">${makeArt(p, 700, 860)}</span>
      <span class="meta">
        <span class="title">${p.title}</span>
        <span class="cat">${p.cat}</span>
      </span>
    `;
    cell.addEventListener('click', () => openLightbox(i, cell));
    grid.appendChild(cell);
  });

  // ---------- filters ----------
  const filterBtns = document.querySelectorAll('.filters button');
  filterBtns.forEach(btn => {
    btn.addEventListener('click', () => {
      filterBtns.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      filterBtns.forEach(b => b.setAttribute('aria-pressed', String(b === btn)));
      const f = btn.dataset.filter;
      document.querySelectorAll('.cell').forEach(cell => {
        cell.classList.toggle('hidden', f !== 'all' && cell.dataset.cat !== f);
      });
    });
  });

  // ---------- lightbox ----------
  const lightbox = document.getElementById('lightbox');
  const lbArt = document.getElementById('lbArt');
  const lbTitle = document.getElementById('lbTitle');
  const lbCounter = document.getElementById('lbCounter');
  let current = 0;

  function visibleIndices(){
    return [...document.querySelectorAll('.cell')]
      .filter(c => !c.classList.contains('hidden'))
      .map(c => Number.parseInt(c.dataset.index, 10));
  }

  let opener = null;
  let previousOverflow = '';

  function openLightbox(index, source){
    if (!Number.isInteger(index) || !photos[index]) return;
    current = index;
    opener = source || document.activeElement;
    previousOverflow = document.body.style.overflow;
    renderLightbox();
    lightbox.classList.add('open');
    lightbox.setAttribute('aria-hidden', 'false');
    document.body.style.overflow = 'hidden';
    document.getElementById('lbClose').focus();
  }

  function renderLightbox(){
    const p = photos[current];
    lbArt.classList.remove('show');
    lbArt.innerHTML = makeArt(p, 900, 1120);
    requestAnimationFrame(() => lbArt.classList.add('show'));
    lbTitle.textContent = p.title;
    const vis = visibleIndices();
    const pos = vis.indexOf(current) + 1;
    lbCounter.textContent = `${frameLabel(current+1)} — ${pos}/${vis.length}`;
  }

  function step(dir){
    const vis = visibleIndices();
    if (!vis.length) return;
    let pos = vis.indexOf(current);
    pos = (pos + dir + vis.length) % vis.length;
    current = vis[pos];
    renderLightbox();
  }

  function closeLightbox(){
    lightbox.classList.remove('open');
    lightbox.setAttribute('aria-hidden', 'true');
    document.body.style.overflow = previousOverflow;
    if (opener && opener.isConnected) opener.focus();
  }

  document.getElementById('lbPrev').addEventListener('click', () => step(-1));
  document.getElementById('lbNext').addEventListener('click', () => step(1));
  document.getElementById('lbClose').addEventListener('click', closeLightbox);
  lightbox.addEventListener('click', (e) => { if (e.target === lightbox) closeLightbox(); });

  document.addEventListener('keydown', (e) => {
    if (!lightbox.classList.contains('open')) return;
    if (e.key === 'ArrowRight') step(1);
    if (e.key === 'ArrowLeft') step(-1);
    if (e.key === 'Escape') closeLightbox();
  });

  // ---------- sprocket holes ----------
  function buildRail(el){
    el.innerHTML = '';
    const count = Math.ceil(window.innerHeight / 28) + 2;
    for (let i = 0; i < count; i++){
      const h = document.createElement('div');
      h.className = 'hole';
      h.style.top = (i * 28 + 7) + 'px';
      el.appendChild(h);
    }
  }
  const railLeft = document.getElementById('railLeft');
  const railRight = document.getElementById('railRight');
  buildRail(railLeft); buildRail(railRight);
  window.addEventListener('resize', () => { buildRail(railLeft); buildRail(railRight); });
</script>
